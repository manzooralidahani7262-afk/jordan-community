from telegram import InlineKeyboardButton, InlineKeyboardMarkup, Update from telegram.ext import Updater, CommandHandler, CallbackContext CHANNEL_LINK = "https://t.me/JordanCommunityUpdates" # اپنا چینل def start(update: Update, context: CallbackContext): if context.args and context.args[0] == "join": keyboard = [[InlineKeyboardButton("✅ Join Jordan Community Updates", url=CHANNEL_LINK)]] update.message.reply_text( "Welcome! Click below to join our Official Channel:", reply_markup=InlineKeyboardMarkup(keyboard) ) updater = Updater("YOUR_BOT_TOKEN", use_context=True) updater.dispatcher.add_handler(CommandHandler("start", start)) updater.start_polling()
